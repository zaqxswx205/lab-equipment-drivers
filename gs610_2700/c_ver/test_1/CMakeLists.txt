cmake_minimum_required(VERSION 3.20)
project(gs610_2700 C)
if(NOT WIN32)
  message(FATAL_ERROR "This version uses Windows native serial and GUI APIs")
endif()
set(CMAKE_C_STANDARD 11)
add_executable(gs610_2700 main.c core.c gui.c)
target_compile_definitions(gs610_2700 PRIVATE _CRT_SECURE_NO_WARNINGS)
if(MSVC)
  target_compile_options(gs610_2700 PRIVATE /W4 /utf-8)
else()
  target_compile_options(gs610_2700 PRIVATE -Wall -Wextra -finput-charset=UTF-8)
endif()
target_link_libraries(gs610_2700 PRIVATE user32 gdi32 comdlg32 comctl32 shell32)
enable_testing()
add_test(NAME planning COMMAND gs610_2700 --self-test)
