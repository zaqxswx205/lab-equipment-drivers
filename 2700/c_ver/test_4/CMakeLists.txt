cmake_minimum_required(VERSION 3.21)
project(Keithley2700Native VERSION 1.0 LANGUAGES C)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "This native Win32 build requires MSVC and the Windows SDK")
endif()
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/W4 /utf-8)
add_compile_definitions(UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS)
add_library(keithley_core STATIC
  src/model.c src/util.c src/serial_port.c src/instrument.c
  src/storage.c src/acquisition.c src/plot.c)
target_include_directories(keithley_core PUBLIC include)
target_link_libraries(keithley_core PUBLIC user32 gdi32 shell32 ole32)
add_executable(Keithley2700 WIN32 src/main.c src/ui.c)
target_link_libraries(Keithley2700 PRIVATE keithley_core comctl32 comdlg32)
target_link_options(Keithley2700 PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/app.manifest")
add_executable(test_core tests/test_core.c)
target_link_libraries(test_core PRIVATE keithley_core)
enable_testing()
add_test(NAME core COMMAND test_core)
add_test(NAME gui_simulation COMMAND Keithley2700 --smoke-test)
