cmake_minimum_required(VERSION 3.16)
project(gs610_sweep LANGUAGES C)
if(NOT WIN32)
    message(FATAL_ERROR "This version uses the Windows native serial API.")
endif()
add_executable(gs610_sweep gs610_sweep.c)
add_executable(gs610_gui WIN32 gs610_gui.c)
target_link_libraries(gs610_gui PRIVATE user32 gdi32 comdlg32 setupapi advapi32 uuid)
set_target_properties(gs610_sweep gs610_gui PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
if(MSVC)
    target_compile_options(gs610_sweep PRIVATE /W4 /utf-8)
    target_compile_options(gs610_gui PRIVATE /W4 /utf-8)
elseif(MINGW)
    target_compile_options(gs610_sweep PRIVATE -Wall -Wextra)
    target_link_options(gs610_sweep PRIVATE -municode)
    target_compile_options(gs610_gui PRIVATE -Wall -Wextra)
    target_link_options(gs610_gui PRIVATE -municode)
endif()
